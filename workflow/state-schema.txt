# State log schema (append-only delta events, key=value payload)
#
# - Stored in output/<topic>_<timestamp>/state.txt.
# - Each committed block contains only one atomic update's changed keys.
# - Current state is the global last-write-wins merge of every committed block.
# - state.current.json and run_status.json are replaceable derived projections.
# - Resume never truncates or restores old state bytes; it appends an event.
# - Legacy blocks remain replayable, and production readers ignore retired
#   review/evaluation values without rewriting the history.
#
# Production fixed-slot contract:
# - Active slots are declared by toc.run_index.SLOT_BY_CODE.
# - Retired review-only slots are p130, p230, p320, p430, p435, p540,
#   p630, p640, p720, p820, p850, and p930.
# - A historical slot.<retired>.status event is data only; it never blocks a
#   current stage and is not emitted in the active slot projection.
#
# Required keys (minimum):
#   timestamp=ISO8601
#   job_id=JOB_YYYY-MM-DD_0001
#   topic=string
#   status=INIT|RESEARCH|STORY|SCRIPT|VIDEO|QA|DONE
#
# Runtime and generation keys:
#   runtime.stage=init|assets|render|done|shorts|...
#   runtime.failure.stage=stage where a terminal runtime failure occurred
#   runtime.failure.phase=grounding|media_generation|validation|runtime
#   runtime.failure.error_kind=timeout|network|runtime_setup|unknown
#   runtime.failure.last_progress_at=ISO8601|unknown
#   runtime.render.status=started|success|failed
#   runtime.target_video_seconds=integer 300..1200
#   runtime.duration_fit.status=passed|failed|skipped
#   runtime.duration_fit.target_seconds=integer
#   runtime.duration_fit.minimum_seconds=integer
#   runtime.duration_fit.actual_seconds=number
#   runtime.duration_fit.ratio=number
#   runtime.duration_fit.measurement_layer=manifest_runtime|final_video
#   runtime.duration_fit.measurement_complete=true|false
#   runtime.duration_fit.spoken_audio_seconds=number
#   runtime.duration_fit.intentional_silence_seconds=number
#   runtime.duration_fit.audio_timeline_seconds=number
#   runtime.duration_fit.video_timeline_seconds=number
#   runtime.duration_fit.video_timeline_source=cuts|render_units|mixed|none
#   runtime.duration_fit.missing_items=[]  # JSON array
#   runtime.duration_fit.invalid_items=[]  # JSON array
#   runtime.duration_fit.note=string
#   runtime.duration_fit.at=ISO8601
#   generation.image_prompt.request_freeze.status=draft|frozen|invalidated
#   generation.image_prompt.request_freeze.request_revision=integer
#   generation.image_prompt.request_freeze.request=image_generation_requests.md
#   generation.image_prompt.request_freeze.snapshot=image_generation_request_snapshot.json
#   generation.image_prompt.request_freeze.frozen_at=ISO8601
#   generation.image_prompt.request_freeze.invalidated_by=string
#   generation.image_prompt.request_freeze.invalidated_at=ISO8601
#   image_generation.status=not_started|in_progress|partial|completed|failed
#   image_generation.started=true|false
#   image_generation.generated_count=non-negative integer
#   image_generation.blocked_by=stage or validation identifier
#   image_generation.block_reason=machine-readable reason
#   orchestration.p600.supervisor.status=pending|running|done|failed|invalidated
#   orchestration.p600.supervisor.invalidated_by=machine-readable reason

# Fixed-slot state keys:
#   slot.pXXX.status=pending|in_progress|done|skipped|blocked|awaiting_approval|failed
#   slot.pXXX.requirement=required|optional
#   slot.pXXX.skip_reason=string
#   slot.pXXX.note=string
#
# Authoring slots retained in production:
#   slot.p410.status=pending|in_progress|done|skipped|blocked|failed
#   slot.p420.status=pending|in_progress|done|skipped|blocked|failed
#   slot.p570.status=pending|in_progress|done|skipped|blocked|failed
#   slot.p680.status=pending|in_progress|done|skipped|blocked|awaiting_approval|failed
#   slot.p750.status=pending|in_progress|done|skipped|blocked|awaiting_approval|failed

# Stage state and grounding keys:
#   stage.<name>.status=pending|in_progress|done|failed|skipped
#   stage.<name>.started_at=ISO8601
#   stage.<name>.finished_at=ISO8601
#   stage.<name>.grounding.status=ready|missing_docs|missing_inputs
#   stage.<name>.grounding.report=logs/grounding/<stage>.json
#   stage.<name>.readset.report=logs/grounding/<stage>.readset.json
#   stage.<name>.playbooks.report=logs/grounding/<stage>.playbooks.json
#   stage.<name>.playbooks.selected_count=non-negative integer
#   stage.<name>.subagent.prompt=logs/grounding/<stage>.subagent_prompt.md
#   stage.<name>.subagent.prompt.generated_at=ISO8601
#
# Canonical stages are research, story, script, asset, scene_implementation,
# narration, video_generation, render, and qa.  Grounding resolves real global
# docs, stage docs, templates, and run inputs; no approval certificate or
# audit-agent result is required.

# Narration runtime keys:
#   runtime.narration.phase=authoring|tts_preview|revising|audio_qa|done
#   runtime.narration.audio_set_hash=sha256:<hex>
#   runtime.narration.timeline_hash=sha256:<hex>
#   runtime.narration.preview_stage=string
#   runtime.narration.preview_note=string
#   artifact.narration_generation_request_snapshot=logs/providers/narration/generation_requests/<stamp>.json

# Artifact and orchestration keys:
#   artifact.video_manifest=output/<topic>_<timestamp>/video_manifest.md
#   artifact.research=output/<topic>_<timestamp>/research.md
#   artifact.story=output/<topic>_<timestamp>/story.md
#   artifact.visual_value=output/<topic>_<timestamp>/visual_value.md
#   artifact.script=output/<topic>_<timestamp>/script.md
#   artifact.asset_inventory=output/<topic>_<timestamp>/asset_inventory.md
#   artifact.asset_plan=output/<topic>_<timestamp>/asset_plan.md
#   artifact.asset_generation_requests=output/<topic>_<timestamp>/asset_generation_requests.md
#   artifact.human_change_log=output/<topic>_<timestamp>/human_change_requests.md
#   artifact.image_generation_requests=output/<topic>_<timestamp>/image_generation_requests.md
#   artifact.image_generation_request_snapshot=output/<topic>_<timestamp>/image_generation_request_snapshot.json
#   artifact.asset_generation_request_snapshot=output/<topic>_<timestamp>/asset_generation_request_snapshot.json
#   artifact.video_generation_requests=output/<topic>_<timestamp>/video_generation_requests.md
#   artifact.grounding.<stage>=output/<topic>_<timestamp>/logs/grounding/<stage>.json
#   artifact.run_index=output/<topic>_<timestamp>/p000_index.md
#   artifact.l2_supervisor_progress=logs/orchestration/l2_supervisor_progress.md
#   orchestration.pXXX.supervisor.progress=logs/orchestration/l2_supervisor_progress.md
#   orchestration.pXXX.supervisor.call_status=invoked|returned|blocked|failed
#   orchestration.pXXX.supervisor.invoked_at=ISO8601
#   orchestration.pXXX.supervisor.last_event_at=ISO8601
#   orchestration.pXXX.supervisor.stop_slot=pXXX
#   orchestration.pXXX.supervisor.status=done|blocked|failed
#   orchestration.pXXX.supervisor.result=logs/orchestration/pXXX.supervisor_result.json
#   orchestration.pXXX.supervisor.finished_at=ISO8601
#   artifact.video=output/<topic>_<timestamp>/video.mp4
#   artifact.video.short.01=output/<topic>_<timestamp>/shorts/short01.mp4

# Legacy compatibility:
# - Existing review.*, eval.*, gate.*, and retired slot keys are preserved in
#   canonical state history when present. They are not active production
#   requirements, progress blockers, score inputs, or approval certificates.
# - Existing state.current.json and p000_index.md files are read as data and
#   may be regenerated without changing state.txt history.
